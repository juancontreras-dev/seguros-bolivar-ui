🚀 Stepper Component

Componente de pasos progresivos para formularios multi-paso, wizards y flujos de trabajo secuenciales. Incluye 3 tipos (Number, Icon, Mix), 2 orientaciones (Horizontal, Vertical) y múltiples estados visuales.

ℹ️ Nota: El Stepper es un componente solo CSS (HTML + CSS puro). No requiere JavaScript, pero puedes añadir lógica personalizada para cambiar estados y navegar entre pasos.

💻 Código de Ejemplo

El Stepper es solo CSS. Para agregar interactividad, actualiza las clases con JavaScript.

HTML Básico

<div class="sb-ui-stepper sb-ui-stepper--horizontal"> <!-- Paso completado --> <div class="sb-ui-stepper__step sb-ui-stepper__step--filled-default"> <div class="sb-ui-stepper__step-circle"> <svg>...check icon...</svg> </div> <div class="sb-ui-stepper__step-label">Paso 1</div> <div class="sb-ui-stepper__step-track"></div> </div> <!-- Paso actual (activo) --> <div class="sb-ui-stepper__step sb-ui-stepper__step--empty-active"> <div class="sb-ui-stepper__step-circle">2</div> <div class="sb-ui-stepper__step-label">Paso 2</div> <div class="sb-ui-stepper__step-track sb-ui-stepper__step-track--dashed"></div> </div> <!-- Paso pendiente --> <div class="sb-ui-stepper__step sb-ui-stepper__step--empty-default"> <div class="sb-ui-stepper__step-circle">3</div> <div class="sb-ui-stepper__step-label">Paso 3</div> </div> </div>

JavaScript para Navegar

function updateStepper(stepperId, currentStep) { const stepper = document.getElementById(stepperId); const steps = stepper.querySelectorAll('.sb-ui-stepper__step'); steps.forEach((step, index) => { const stepNumber = index + 1; const circle = step.querySelector('.sb-ui-stepper__step-circle'); const track = step.querySelector('.sb-ui-stepper__step-track'); // Remover todas las clases de estado step.classList.remove( 'sb-ui-stepper__step--empty-default', 'sb-ui-stepper__step--empty-active', 'sb-ui-stepper__step--filled-default' ); if (track) { track.classList.remove('sb-ui-stepper__step-track--dashed'); } // Aplicar estado según paso actual if (stepNumber < currentStep) { // Paso completado step.classList.add('sb-ui-stepper__step--filled-default'); circle.innerHTML = '<svg>...check...</svg>'; } else if (stepNumber === currentStep) { // Paso actual step.classList.add('sb-ui-stepper__step--empty-active'); circle.textContent = stepNumber; if (track) track.classList.add('sb-ui-stepper__step-track--dashed'); } else { // Paso pendiente step.classList.add('sb-ui-stepper__step--empty-default'); circle.textContent = stepNumber; if (track) track.classList.add('sb-ui-stepper__step-track--dashed'); } }); }
